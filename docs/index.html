Quick Start

Deploy EvolvixOS in minutes

# Clone the repository git clone https://github.com/Protremix/EvolvixOS.git cd EvolvixOS # Deploy (requires Ubuntu 22.04+) chmod +x gpu_deploy.sh ./gpu_deploy.sh # Or manual setup pip3 install -r requirements.txt python3 models/model_api.py & # Model API on :5010 python3 auth/auth_api.py & # Auth API on :5000 python3 dashboard/server.py & # Dashboard on :8080 # Configure (optional) export GROQ_API_KEY="your-groq-key" export GEMINI_API_KEY="your-gemini-key" export KIMI_API_KEY="your-kimi-key" export TENCENTCLOUD_SECRET_ID="your-tc-id" export TENCENTCLOUD_SECRET_KEY="your-tc-key"

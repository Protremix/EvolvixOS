from evolvixos import EvolvixOS

client = EvolvixOS(api_key="YOUR_API_KEY")

# Chat with any model
resp = client.chat("Write a haiku about code")
print(resp["response"])

# Stream responses
for chunk in client.stream("Tell me a story"):
    print(chunk, end="", flush=True)

# Create entities
client.entities.create("Task", {
    "title": {"type": "string"},
    "status": {"type": "string", "enum": ["todo", "done"]}
})

# Create records
client.entities.records("Task").create({"title": "Ship it"})

# Create AI agents
client.agents.create("reviewer", system_prompt="You are a code reviewer")
review = client.agents.chat("reviewer", "Review this function")
print(review["response"])

import { EvolvixOS } from 'evolvixos';

const client = new EvolvixOS('YOUR_API_KEY');

// Chat with any model
const resp = await client.chat('Write a haiku about code');
console.log(resp.response);

// Stream responses
for await (const chunk of client.stream('Tell me a story')) {
  process.stdout.write(chunk);
}

// Create entities
await client.entities.create('Task', {
  title: { type: 'string' },
  status: { type: 'string', enum: ['todo', 'done'] }
});

// Create AI agents
await client.agents.create('reviewer', 'You are a code reviewer');
const review = await client.agents.chat('reviewer', 'Review this code');

Self-Host

Run EvolvixOS on your own infrastructure — Docker Compose or Kubernetes.

Docker Compose
Kubernetes
git clone https://github.com/Protremix/EvolvixOS.git
cd EvolvixOS
cp .env.example .env  # Edit secrets
docker compose up -d  # Done!

Includes PostgreSQL, Ollama (local models), and optional Nginx. All you need is a free Groq or NVIDIA key for cloud models.

helm install evolvixos ./helm/evolvixos \
  --set env.JWT_SECRET="your-secret" \
  --set env.GROQ_API_KEY="your-key"

Includes PostgreSQL StatefulSet, Ollama with PVC, Ingress, and autoscaling. GPU support for Ollama is configurable.
